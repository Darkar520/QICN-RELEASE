\section{Formal Setup}

\subsection{Review of Identity Framework}

We assume the identity framework of \cite{paper1}. Let $(S_t, \pi_{t+1 \to t})_{t \in \mathbb{N}}$ be a projective system of observable state spaces. The identity object is the inverse limit:
\begin{equation}
    \mathcal{I} := \varprojlim S_t = \left\{ (x_t)_t \in \prod_t S_t \;\Big|\; \pi_{t+1 \to t}(x_{t+1}) = x_t \;\; \forall t \right\}.
\end{equation}

The sectorial deformation resistance $M_\Omega$ quantifies resistance to identity deformation. We recall that:
\begin{itemize}[leftmargin=*]
    \item $M_\Omega = 0$: zero deformation-cost infimum;
    \item $0 < M_\Omega < \infty$: deformable persistence (finite rigidity);
    \item $M_\Omega = +\infty$: closed--rigid (CCR relative to the declared perturbation class).
\end{itemize}

\subsection{Abstract Structural Readout Space}

\begin{definition}[Structural Readout Space]\label{def:phenspace}
A structural readout space is a nonempty set $\mathcal{E}$ whose elements are called \textit{phenomenological configurations}. No topology, metric, or algebraic structure is assumed a priori.
\end{definition}

\begin{remark}
We do not claim that $\mathcal{E}$ represents ``conscious states'' or ``qualia''. It is an abstract mathematical space whose interpretation is not specified by the formalism. The space $\mathcal{E}$ is introduced purely as the codomain of a structural assignment; its ontological status is deliberately left open.
\end{remark}

\subsubsection{Why a Separate Space?}

One might ask: why not use $\mathcal{I}$ itself as the structural readout space? The answer is structural:

\begin{proposition}[Endomorphisms Preserving All Canonical Projections]
If $F:\mathcal I\to\mathcal I$ satisfies $p_t\circ F=p_t$ for every $t$, then $F=\operatorname{Id}$. This does not exclude endomorphisms with nonidentity coordinate actions.
\end{proposition}

\begin{proof}
For each $x$, $F(x)$ and $x$ have identical coordinates, hence are the same inverse-limit element. For contrast, on binary prefix spaces, bitwise complement defines an endomorphism induced by complement on every prefix; it commutes with the bonding maps but does not preserve each coordinate value. A separate codomain is a modeling choice, not a logical necessity for nonidentity maps.
\end{proof}

A separate codomain permits independent structural readouts; it is not forced by the universal property alone.

\subsection{Structural Assignment}

\begin{definition}[Structural Assignment]\label{def:phenassign}
A structural assignment is a family of total maps on the declared observable state spaces:
\begin{equation}
    \phi_t: S_t \to \mathcal{E}
\end{equation}
indexed by $t \in \mathbb{N}$.
\end{definition}

\begin{definition}[Induced Limit Assignment]\label{def:limitassign}
Given a structural assignment $\{\phi_t\}$, the induced limit assignment, whenever it exists, is the map:
\begin{equation}
    \Phi: \mathcal{I} \to \mathcal{E}, \quad \Phi((x_t)_t) := \lim_{t \to \infty} \phi_t(x_t),
\end{equation}
The convergence statements below use the complete metric codomain of Hypothesis~\ref{hyp:H3-metric}. Partial readouts may instead be studied on a specified common domain, with all projections and factorizations restricted to that domain.
\end{definition}

\subsubsection{Uniqueness of the Limit Assignment}

\begin{proposition}[Uniqueness and Uniform Error of the Limit Assignment]\label{prop:unique-phi}
Under E1, each trajectory has a unique limit assignment and
\[
d_{\mathcal E}\bigl(\Phi(x),\phi_n(p_nx)\bigr)\leq A_n
\quad\text{for all }x\in\mathcal I.
\]
\end{proposition}

\begin{proof}
For $m>n$, the triangle inequality bounds the distance between the $m$th and $n$th readings by $\sum_{t=n}^{m-1}a_t$. Thus each sequence is Cauchy. Completeness supplies its limit, uniqueness follows from metric separation, and passage to the limit gives the bound. Since $A_n\to0$ independently of $x$, convergence is uniform.
\end{proof}

\subsection{Axioms for Phenomenological Coherence}

We declare a controlled assignment model. E0 is a nontriviality convention, E1 supplies convergent readouts, and E2 is a separately checked non-factorization requirement.

\begin{axiom}[E0: Non-Triviality]\label{ax:e0}
The structural readout space satisfies $|\mathcal{E}| \geq 2$, i.e., there exist at least two distinct phenomenological configurations.
\end{axiom}

\begin{axiom}[E1: Controlled Asymptotic Compatibility]\label{ax:e1}
In the complete metric codomain of Hypothesis~\ref{hyp:H3-metric}, there are numbers $a_t\geq0$, independent of the trajectory, with $\sum_{t=0}^{\infty}a_t<\infty$, such that
\[
d_{\mathcal E}\bigl(\phi_{t+1}(p_{t+1}x),\phi_t(p_tx)\bigr)\leq a_t
\qquad(x\in\mathcal I,\ t\geq0).
\]
Here $p_t:\mathcal I\to S_t$ is the canonical projection. Define the tail budget $A_n=\sum_{t=n}^{\infty}a_t$. Exact equality is the special case $a_t=0$, not a requirement of E1.
\end{axiom}

\begin{axiom}[E2: Non-Locality]\label{ax:e2}
There exists no function $f: S_t \to \mathcal{E}$ (for any fixed $t$) such that for all trajectories $(x_t)_t \in \mathcal{I}$:
\begin{equation}
    \Phi((x_t)_t) = f(x_t).
\end{equation}
\end{axiom}

\subsection{Status of the Assignment Conditions}

The following results separate sufficient assumptions, a non-factorization requirement and their logical dependencies.

\subsubsection{Nontriviality of the Codomain}

\begin{theorem}[E0 is Necessary]\label{thm:e0-necessary}
If $|\mathcal{E}| = 1$, then every structural assignment is constant and no phenomenological classification is possible.
\end{theorem}

\begin{proof}
If $\mathcal E=\{e_0\}$, any total assignment takes the value $e_0$ on every trajectory. It distinguishes no trajectories and factors through every canonical projection. It therefore cannot meet E2 on a nonempty domain.
\end{proof}

\subsubsection{Controlled Convergence}

\begin{theorem}[Controlled Compatibility Is Sufficient]\label{thm:e1-necessary}
E1 guarantees existence of the limit with the error bound in Proposition~\ref{prop:unique-phi}. Absolute summability is sufficient, not necessary for an arbitrary convergent family.
\end{theorem}

\begin{proof}
Existence and the bound were proved above. To disprove necessity in general, take readouts constant in the trajectory with values $s_n=\sum_{k=1}^{n}(-1)^{k+1}/k$ in $[0,1]$. The even partial sums increase, the odd partial sums decrease, and their difference tends to zero, so they have a common limit. Their successive absolute increments form the divergent harmonic series; no summable uniform majorant exists. This example is a control for the necessity claim, not a realization of E2.
\end{proof}

\subsubsection{Finite-Cut Non-Factorization}

\begin{theorem}[Non-Factorization Is an Independent Requirement]\label{thm:e2-necessary}
E2 excludes readouts determined by a single canonical finite cut. The inverse-limit construction and E1 alone do not imply E2.
\end{theorem}

\begin{proof}
Take any nonempty inverse limit and a constant family $\phi_t=e_0$ in a codomain containing at least two points. E1 holds with zero tail, but $\Phi=e_0$ factors through every projection. Thus E2 must be checked separately. The nonempty construction in Theorem~\ref{thm:ternary-regime-realization} checks it without assuming it.
\end{proof}

\subsection{Logical Dependencies and Realizability}

\begin{theorem}[Logical Dependencies of the Assignment Conditions]\label{thm:minimality}
On a nonempty trajectory domain, E2 implies E0. E1 does not imply E2. There exist assignments satisfying E0, the revised E1 and E2 simultaneously. Consequently these three conditions are not an independent minimal axiom basis as originally stated.
\end{theorem}

\begin{proof}
If the codomain were a singleton, every assignment would be constant and factor through $p_0$, contradicting E2. The preceding constant-family control proves the second assertion. Theorem~\ref{thm:ternary-regime-realization} proves the third by construction. E0 is retained as an explicit convention, although E2 entails it in the nonempty case.
\end{proof}

\subsection{Rejected Alternatives}

We briefly discuss axiom candidates that were considered and rejected.

\subsubsection{Rejected: Continuity Axiom}
One might require $\Phi$ to be continuous. This is not required by the chosen model because:
\begin{itemize}[leftmargin=*]
    \item The complete-metric model permits discontinuous finite readouts; continuity is a property of the regular-readout subclass, not part of the unrestricted assignment condition;
    \item Continuity is derived from controlled approximation by continuous finite readouts; a scalar CCR label alone is insufficient.
\end{itemize}

\subsubsection{Rejected: Surjectivity Axiom}
One might require $\Phi$ to be surjective. This is not required by the chosen model because:
\begin{itemize}[leftmargin=*]
    \item A readout need not exhaust a codomain chosen to include other regimes;
    \item Surjectivity is a strong structural claim not derivable from identity constraints.
\end{itemize}

\subsubsection{Rejected: Injectivity Axiom}
One might require $\Phi$ to be injective. This is not required by the chosen model because:
\begin{itemize}[leftmargin=*]
    \item Many trajectories may share phenomenological configurations;
    \item Injectivity is not imposed on the full assignment class; it is a consequence in the concrete coercive realization. An injective map alone is not necessarily a topological embedding.
\end{itemize}

\subsection{Exact Compatibility Obstruction}

\begin{proposition}[Exact Compatibility Forces Finite-Cut Factorization]\label{prop:exact-e1-obstruction}
On a nonempty trajectory domain, if $\phi_{t+1}(p_{t+1}x)=\phi_t(p_tx)$ for every $x,t$, then the limit assignment satisfies $\Phi=\phi_0\circ p_0$. It cannot satisfy E2. The controlled E1 above replaces that exact equality, rather than assuming it together with E2.
\end{proposition}

\begin{proof}
Induction makes all readings equal to the first. Their limit is therefore $\phi_0(p_0x)$. For partial readouts with a declared common domain, the same argument holds on that domain. If a total map on $S_0$ is required, extend the defined first-cut values arbitrarily outside the projected common domain, using any fixed element of the nonempty codomain. Thus partiality alone does not remove the obstruction.
\end{proof}

\subsection{A Nonempty Controlled Assignment Model}

\begin{theorem}[Ternary Realization of E0--E2]\label{thm:ternary-regime-realization}
Let $S_t=\{0,1\}^{t+1}$ with bonding maps that delete the last bit. Identify its inverse limit with $\mathcal I=\{0,1\}^{\mathbb N}$ and let $\mathcal E=[0,1]$. Define
\[
\phi_t(b_0,\ldots,b_t)=\sum_{j=0}^t\frac{2b_j}{3^{j+1}},
\qquad
\Phi(b)=\sum_{j=0}^{\infty}\frac{2b_j}{3^{j+1}}.
\]
These assignments satisfy E0, controlled E1, and E2, with
\[
a_t=\frac{2}{3^{t+2}},\qquad
A_t=\frac{1}{3^{t+1}},\qquad
\sup_b|\Phi(b)-\phi_t(p_tb)|=A_t.
\]
For distinct histories first differing at position $m$, put $d_{\mathcal I}(b,c)=3^{-(m+1)}$, and put $d_{\mathcal I}(b,b)=0$. Then
\[
d_{\mathcal I}(b,c)\leq|\Phi(b)-\Phi(c)|\leq3d_{\mathcal I}(b,c).
\]
The domain is compact and complete, and the readout is injective and bi-Lipschitz onto its image. Thus a global lower bound is realized here rather than inserted without a witness.
\end{theorem}

\begin{proof}
Every infinite bit sequence defines a compatible prefix history, so the domain is nonempty. Geometric summation puts every value in $[0,1]$ and gives the increment and tail bounds. The all-one tail attains the tail bound. For any fixed $t$, a zero history and a history with its sole one at position $t+1$ have the same prefix but images differing by $2/3^{t+2}>0$. No function of $p_t$ determines $\Phi$, proving E2 for every finite cut.

At the first differing position $m$, the principal difference has magnitude $2/3^{m+1}$; the remaining tail has magnitude at most $1/3^{m+1}$. Subtraction and addition give the two bounds. This metric is the weighted supremum of the discrete prefix metrics with weights $3^{-(t+1)}$; bounded prefix diameters ensure finiteness. A Cauchy sequence for $d_{\mathcal I}$ eventually stabilizes each finite prefix, which defines its limit. Prefix cylinders give finite covers of arbitrarily small diameter; the space is totally bounded and complete, hence compact. The two bounds establish injectivity and quantitative continuity. Compactness and continuity give properness into the Hausdorff codomain, and a later-bit flip in every neighborhood gives local non-collapse. Thus this realization also supplies all four clauses of the stated readout regularity hypothesis. A scalar real-valued readout still encodes the complete sequence: E2 excludes the specified canonical finite cuts, not every possible coding or every finite-dimensional representation.
\end{proof}

\begin{theorem}[Sharp Finite-Cut Approximation Error]\label{thm:finite-cut-minimax}
For the ternary realization and an arbitrary decoder $g:S_t\to\mathbb R$, let
\[
R_t(g)=\sup_{b\in\mathcal I}|\Phi(b)-g(p_tb)|.
\]
Then the exact minimax value is
\[
\inf_g R_t(g)=\frac{1}{2\,3^{t+1}}.
\]
It is attained by $g_\ast(s)=\phi_t(s)+1/(2\,3^{t+1})$.
\end{theorem}

\begin{proof}
For a fixed prefix $s$, the all-zero and all-one suffixes give endpoints $\phi_t(s)$ and $\phi_t(s)+A_t$. For any number $g(s)$, the triangle inequality forces at least one endpoint error to be at least $A_t/2$. Conversely all readout values in that fiber lie between those endpoints, so their midpoint has error at most $A_t/2$. Both endpoints belong to the domain, establishing the exact optimum and attainment. The bound is positive for each fixed finite budget and tends to zero as $t$ increases; it gives no positive uniform error floor across increasing resolutions.
\end{proof}

\begin{proposition}[Sharp Error for Arbitrary Binary Codes]\label{prop:ternary-arbitrary-code}
For the ternary assignment above, let an encoder choose at most $2^B$ states as an arbitrary function of the whole history, and let each state decode to any real number. The minimum worst-case absolute error is $1/(2\cdot3^B)$ for every integer $B\geq0$.
\end{proposition}
\begin{proof}
The cylinders of depth $B$ have intervals of length $3^{-B}$ between their two attainable endpoints. Different intervals have gaps at least $3^{-B}$: after the first differing bit $j\leq B$, the maximum opposing tail is $3^{-j}$. Thus all $2^{B+1}$ endpoints, including the depth-zero endpoints $0$ and $1$, are pairwise separated by at least $3^{-B}$. Two share a code and the triangle inequality forces error at least $3^{-B}/2$. Encoding the $B$-bit prefix and decoding to its interval midpoint attains the bound.
\end{proof}

\subsection{Common Readout on Two Coherently Transported Towers}

\begin{theorem}[Joint Realization of Transport and Controlled Readouts]\label{thm:joint-transport-readout}
Use circle-prefix stages $S_t=\mathbb T\times\{0,1\}^t$, stage metric $\max(\rho,\mathbf{1}_{\text{different prefix}})$, and weights $w_t=3^{-t}$, where $\rho$ is circle distance with circumference one. Take the usual truncation tower $I_0$ and its variant $I_1$ with angle shift $-a$ at the single bonding level $k$, for $0<a<1/2$. Write $\beta=3^{-(k+1)}$. In the common weighted history space put $Z=I_0\cup I_1$. Let the complete codomain be $\mathcal E=\mathbb T^2\times[0,1]$ with the maximum of the first circle distance, $\beta$ times the second circle distance, and the absolute scalar difference. Define
\[
 F(x)=\left(\theta_0(x),\theta_{k+1}(x),\sum_{j\geq1}2b_j(x)3^{-j}\right).
\]
Then $d_w(x,y)\leq d_{\mathcal E}(F(x),F(y))\leq3d_w(x,y)$ on all of $Z$. On each tower the restriction satisfies E0--E2 with $a_t=2\cdot3^{-(t+1)}$ and uniform tail $A_t=3^{-t}$. The coordinate homeomorphism $H:I_0\to I_1$ shifts angles at stages $t>k$ by $a$. With projection-defect cost $E_w=3^{-k}a$, one has
\[
 d_w(x,Hx)=d_{\mathcal E}(F(x),F(Hx))=\beta a,\qquad
 d_H(I_0,I_1)=\frac{\beta a}{1+\beta}.
\]
\end{theorem}
\begin{proof}
In each tower the angle is constant on the two blocks $t\le k$ and $t>k$. Thus the angular part of the history distance is exactly the maximum of the two weighted angular distances in the codomain. The binary part is $3^{-j}$ at first differing bit $j$. The leading ternary term and its tail give lower and upper constants $1$ and $3$ for that part. Each tower is closed in the compact metric product, so their union is complete.

For a declared tower $s\in\{0,1\}$ the finite observation $(\theta_t,b_1,\ldots,b_t)$ gives the initial angle $\theta=\theta_t-sa\mathbf 1_{t>k}$. The stage readout is $(\theta,\theta+sa,\sum_{j=1}^t2b_j3^{-j})$. The phase rule uses the declared bonding family, and on both towers the successive readout increment is bounded by $a_t$, with exactly the stated uniform tail. A later-bit flip proves E2, since both phase coordinates stay fixed while the scalar changes. The codomain is nontrivial, proving E0. Within either fixed-angle fiber the sharp binary-code result applies to the scalar readout.

The coordinate squares commute because the sole change of phase at stage $k+1$ cancels the shift $-a$ in its bonding map. The coordinatewise inverse shifts by $-a$ on the later block. For fixed bits, $F$ measures the shift $\beta a$ exactly. In the comparison of sets, a pair at distance $r$ has $a\leq r+r/\beta$ by the triangle inequality at stages $0$ and $k+1$. Equal bit sequences and an initial phase offset $-\beta a/(1+\beta)$ for the second history attain this bound in both directions.
\end{proof}


\subsection{Controls Separating Convergence, Continuity, and Invariance}

\begin{proposition}[Setwise Invariance Does Not Supply Continuity]\label{prop:readout-continuity-control}
Controlled E1 and E2 can hold for a discontinuous assignment, even if the identity domain is fixed under every declared intervention.
\end{proposition}

\begin{proof}
Take $S_t=[0,1]\times\{0,1\}^{t+1}$, with prefixes on the bit coordinates, and readouts
\[
g_t(s,b)=\tfrac12\mathbf1_{\{s>0\}}+\sum_{j=0}^t\frac{b_j}{3^{j+1}}.
\]
The increments are bounded by $3^{-(t+2)}$, so they satisfy controlled E1 in $[0,1]$. Later bits prove E2 as above. With all bits zero, values at $s=0$ and $s=1/n$ differ by $1/2$ although their product-metric distance tends to zero. The limit is discontinuous. Let the declared intervention family consist only of the identity: the domain is setwise fixed, which does not alter the counterexample. This is a control for the asserted inference from invariance to continuity; no claim of full CCR or operational-criterion membership is made for it.
\end{proof}

\begin{proposition}[Setwise and Pointwise Transport Are Distinct]\label{prop:transport-control}
Preserving an identity domain as a set does not guarantee equal values of a common readout on transported states.
\end{proposition}

\begin{proof}
On $Z=\{0,1\}$, let $h(0)=1$, $h(1)=0$ and $F(x)=x$. The transport preserves $Z$ exactly but changes both readout values. Thus setwise rigidity cannot replace the pointwise or readout-natural comparison hypotheses used below.
\end{proof}

\section{Compatibility with Identity Regimes}

\subsection{Induced Constraints from Identity Rigidity}

Let $(\mathcal{I}, M_\Omega)$ be an identity object with sectorial deformation resistance $M_\Omega$. We now study how $M_\Omega$ constrains the set of admissible structural assignments.

\begin{definition}[Phenomenological Compatibility Class]\label{def:compat-class}
For a declared identity domain and codomain, $\mathcal P(M_\Omega)$ consists of maps $\Phi:\mathcal I\to\mathcal E$ that are limits of some family $\{\phi_t\}$ satisfying E1, with E0 and E2 also satisfied. The readout family and its tail control are part of the realization data. The notation retains the mass index for comparisons, but no mass-dependent restriction follows from E0--E2 alone.
\end{definition}

\begin{remark}
The compatibility class records choices of domain, codomain and readout family. A numerical mass label does not by itself determine those choices or their transport under perturbations.
\end{remark}

\subsection{Metric Structure for Quantitative Comparisons}

Quantitative comparisons use a declared metric. The following statements identify the role and limits of that choice.

\begin{hypothesis}[H3: Metric on $\mathcal{E}$]\label{hyp:H3-metric}
The structural readout space $\mathcal{E}$ admits a metric $d_\mathcal{E}$ such that $(\mathcal{E}, d_\mathcal{E})$ is a complete metric space.
\end{hypothesis}

\subsubsection{Role of the Metric}

\begin{theorem}[Metric Data Required by Quantitative Formulas]\label{thm:h3-necessary}
The distance-based fragmentation and loss functionals used here require a declared metric, or a pseudometric with a stated quotient convention. Topological continuity can be defined without a metric. Completeness is used for the Cauchy-limit guarantee of E1.
\end{theorem}

\begin{proof}
Each quantitative functional explicitly contains $d_{\mathcal E}$, so it is undefined until that datum is supplied. Continuity can alternatively be defined through inverse images of open sets. A pseudometric can identify zero-distance states by a quotient, but that changes the domain of class labels and must be specified. These facts justify the chosen quantitative model without claiming that all alternatives are impossible.
\end{proof}

\subsubsection{Completeness as a Limit Guarantee}

\begin{proposition}[Completeness and Existence of Limits]\label{prop:complete-necessary}
Without completeness, the conclusion that every E1-type Cauchy family has its limit in the declared codomain can fail. Completeness is not necessary for a family whose limit is already known to belong to that codomain.
\end{proposition}

\begin{proof}
In $\mathbb Q\cap[0,2]$ with its usual metric, use dyadic rational approximations $q_n=2^{-n}\lfloor2^n\sqrt2\rfloor$ as constant-in-trajectory readings. The increments are bounded by $2^{-n}$, a summable sequence, but their real limit $\sqrt2$ is outside the codomain. This isolates the role of completeness when supplying a general existence guarantee; it makes no assertion that a convergent constant family needs completeness.
\end{proof}

\subsubsection{Alternatives Considered and Rejected}

\textbf{Alternative 1: Topology without Metric.}
One might use a general topological space. This is not adopted in the present quantitative model because:
\begin{itemize}[leftmargin=*]
    \item Fragmentation requires quantitative distance, not just open sets;
    \item Loss capacity requires a sup over distances;
    \item The downstream instability analysis completed in \textit{Null-Regime Instability} requires metric quotients.
\end{itemize}

\textbf{Alternative 2: Pseudometric.}
One might allow $d_\mathcal{E}(e_1, e_2) = 0$ for $e_1 \neq e_2$. This is not adopted in the present quantitative model because:
\begin{itemize}[leftmargin=*]
    \item It would conflate distinct phenomenological configurations;
    \item Cardinality and metric separation are different requirements; zero-distance classes would require a declared quotient before distinguishability is counted.
\end{itemize}

\textbf{Alternative 3: Non-Complete Metric.}
This is rejected by Proposition~\ref{prop:complete-necessary}.

\subsection{Phenomenological Continuity}

\begin{definition}[Phenomenological Continuity]\label{def:phen-continuity}
A structural assignment $\Phi: \mathcal{I} \to \mathcal{E}$ is \textit{structurally continuous} if for every sequence $(x^{(n)})_n$ in $\mathcal{I}$ converging in the $d_w$ metric:
\begin{equation}
    d_w(x^{(n)}, x) \to 0 \implies d_\mathcal{E}(\Phi(x^{(n)}), \Phi(x)) \to 0.
\end{equation}
\end{definition}

\begin{remark}
Structural continuity is derived under the finite-readout conditions of Theorem~\ref{thm:forced-continuity}. It is distinct from invariance under an external intervention.
\end{remark}

\subsection{Phenomenological Fragmentation}

\begin{definition}[Phenomenological Fragmentation]\label{def:fragmentation}
A structural assignment $\Phi$ is \textit{fragmentable} if there exists a perturbation $\{\epsilon_t\}$ with finite weighted energy such that:
\begin{equation}
    d_\mathcal{E}(\Phi(x), \tilde{\Phi}(\tilde{x})) > 0
\end{equation}
for corresponding identity trajectories $x \in \mathcal{I}$ and $\tilde{x} \in \tilde{\mathcal{I}}$.
\end{definition}

\begin{remark}
Fragmentation captures the vulnerability of phenomenological structure to perturbations. It is the phenomenological analog of identity deformability.
\end{remark}

\subsection{Relationship Between Metrics}

\begin{proposition}[Continuity from Controlled Finite Readouts]\label{prop:metric-compat}
Write $g_n=\phi_n\circ p_n$. Under E1,
\[
d_{\mathcal E}(\Phi(x),\Phi(y))\leq2A_n+d_{\mathcal E}(g_n(x),g_n(y)).
\]
If every $g_n$ is continuous, then $\Phi$ is continuous. If every $g_n$ is $L$-Lipschitz with a common $L$, then $\Phi$ is $L$-Lipschitz.
\end{proposition}

\begin{proof}
Insert $g_n(x)$ and $g_n(y)$ and apply the uniform error bound. For continuity, first make $2A_n$ smaller than half the prescribed error, then use continuity of that fixed $g_n$. For the Lipschitz conclusion use the same bound for all $n$ and let $n\to\infty$.
\end{proof}

\subsection{Closure of Section 3}
The quantitative formulas use a declared complete metric codomain. Continuous finite readouts and controlled tails supply a continuity theorem. Cross-system comparisons additionally require a declared common domain and transport; neither mass labels nor distances between sets substitute for those data.

\section{Impossibility Theorems}

\subsection{\texorpdfstring{$\Phi$}{Phi}-Regularity Hypothesis}

Before proving the main impossibility results, we isolate the global lower-bound regularity condition used by the fragmentation argument. The condition is not derived from properness, local Lipschitz regularity, and non-collapse alone; it is carried as an explicit hypothesis.

\begin{hypothesis}[$\Phi$-Regularity Lower Bound]\label{hyp:phi-regularity}
Let $(\mathcal{I}, d_w)$ be the identitary metric space from \textit{Rigid Identity} and let $(\mathcal{E}, d_\mathcal{E})$ be the abstract phenomenological metric space. A structural assignment $\Phi: \mathcal{I} \to \mathcal{E}$ is said to satisfy $\Phi$-regularity when it satisfies:
\begin{enumerate}[leftmargin=*]
    \item (Properness) For every compact $K \subset \mathcal{E}$, $\Phi^{-1}(K)$ is compact in $\mathcal{I}$.
    \item (Local Lipschitz regularity) For every $x \in \mathcal{I}$, there exist $\varepsilon > 0$ and $L_x > 0$ such that for all $y, z \in B_\varepsilon(x)$:
    \begin{equation}
        d_\mathcal{E}(\Phi(y), \Phi(z)) \leq L_x \cdot d_w(y, z).
    \end{equation}
    \item (Non-collapse) For every $x \in \mathcal{I}$ and every $r > 0$, the restriction of $\Phi$ to $B_r(x)$ is not constant.
    \item (Global lower Lipschitz bound) There exists a constant $C>0$ such that for all $x,y\in\mathcal{I}$,
    \begin{equation}
        d_\mathcal{E}(\Phi(x),\Phi(y)) \geq C\cdot d_w(x,y).
    \end{equation}
\end{enumerate}
\end{hypothesis}

\begin{caveat}[Status of $\Phi$-regularity]
The fourth clause is an additional structural assumption, not a consequence of the first three clauses. The counterexample in Appendix~\ref{app:phi-regularity-counterexample} shows that properness, local Lipschitz regularity, and non-collapse do not imply any uniform lower Lipschitz constant.
\end{caveat}

\begin{hypothesis}[Audited Perturbation Comparisons]\label{hyp:audited-comparison}
Declare a finite-cost intervention class $\mathcal V$, a common metric space $(Z,d_Z)$ containing the unperturbed and perturbed identity domains, and transports $h_\varepsilon:\mathcal I\to\mathcal I_\varepsilon\subseteq Z$. Readouts are restrictions of one declared map $F:Z\to\mathcal E$: $\Phi=F|_{\mathcal I}$ and $\Phi_\varepsilon=F|_{\mathcal I_\varepsilon}$.
The following branches are alternatives, used separately. The positive-displacement branch has some $\varepsilon\in\mathcal V$ and $x\in\mathcal I$ with $d_Z(x,h_\varepsilon x)>0$. The zero-displacement branch has $h_\varepsilon x=x$ for every $\varepsilon\in\mathcal V$ and $x\in\mathcal I$. Any use of the lower Lipschitz bound in a comparison requires that bound for $F$ on the common comparison domain, not merely within one identity object.
\end{hypothesis}

\begin{remark}[Burden linking rigidity and readout comparisons]
The scalar $M_\Omega$ does not itself supply the transports or a common readout. Relating its finite and infinite regimes to these two branches is a separate model obligation. Setwise invariance does not imply pointwise fixed transport, and a lower bound within one domain does not compare unrelated assignments on different domains.
\end{remark}

\subsection{Fragmentation Theorem}

\begin{theorem}[Fragmentation under an Audited Displacement]\label{thm:fragmentation}
Let an assignment satisfy E0--E2. Assume the positive-displacement branch of Hypothesis~\ref{hyp:audited-comparison} and a lower Lipschitz constant $C>0$ for $F$ on its common comparison domain. Then, for the witnessing pair,
\[
d_{\mathcal E}(\Phi(x),\Phi_\varepsilon(h_\varepsilon x))\geq C d_Z(x,h_\varepsilon x)>0.
\]
Thus the assignment is fragmentable. This applies to a finite-$M_\Omega$ model when that model supplies the stated comparison data.
\end{theorem}

\begin{proof}
Both arguments lie in the declared domain of $F$. Its lower bound gives the inequality, and the intervention has finite cost by membership in $\mathcal V$. E2 is not used to infer that every deformation changes a readout: the lower bound and the positive witness perform that work.
\end{proof}

\begin{corollary}[Fragmentability and Continuity Are Distinct]
The positive-displacement conclusion does not establish a topological discontinuity. A continuous assignment may be fragmentable under a state-changing intervention.
\end{corollary}

\begin{proof}
The ternary realization has a continuous assignment. Declare a finite-cost intervention that flips one bit while retaining the common readout. Its value changes by a positive amount, so the assignment is fragmentable in the stated sense while remaining continuous. This does not assert a full CCR realization.
\end{proof}

\subsection{Forced Continuity Theorem}

\begin{theorem}[Continuity and Conditional Perturbation Invariance]\label{thm:forced-continuity}
Under E1, if each finite readout $g_n=\phi_n\circ p_n$ is continuous, then $\Phi$ is continuous, irrespective of $M_\Omega$. Under the zero-displacement branch of Hypothesis~\ref{hyp:audited-comparison},
\[
\Phi_\varepsilon(h_\varepsilon x)=\Phi(x)
\qquad(\varepsilon\in\mathcal V,\ x\in\mathcal I).
\]
For a CCR model the invariance conclusion therefore holds when its finite-cost interventions admit that branch. CCR alone is not a proof of continuity or of readout naturality.
\end{theorem}

\begin{proof}
Continuity follows from Proposition~\ref{prop:metric-compat}. For invariance use the common readout and the fixed transport: $\Phi_\varepsilon(h_\varepsilon x)=F(h_\varepsilon x)=F(x)=\Phi(x)$. Neither argument identifies invariance under external perturbation with continuity along convergent states.
\end{proof}

\begin{corollary}[Invariance on the Audited Zero-Displacement Branch]
A CCR model satisfying the zero-displacement comparison hypothesis has invariant assignment values under all interventions in its declared finite-cost class.
\end{corollary}

\begin{proof}
Apply the common-readout equality in Theorem~\ref{thm:forced-continuity} for each declared intervention.
\end{proof}

\subsection{Dichotomy Theorem}

\begin{theorem}[Conditional Classification by Audited Branches]\label{thm:dichotomy}
Suppose a model supplies the positive-displacement branch and lower bound whenever $M_\Omega<\infty$, and supplies continuous finite readouts and the zero-displacement branch whenever $M_\Omega=+\infty$. Then its finite regime is fragmentable and its infinite regime has a continuous, perturbation-invariant assignment. The mass cases are exclusive; fragmentability and topological continuity need not be exclusive properties.
\end{theorem}

\begin{proof}
Apply Theorems~\ref{thm:fragmentation} and~\ref{thm:forced-continuity} with their respective comparison hypotheses. Exclusivity of the cases follows from the numerical alternatives for $M_\Omega$, not from a false identification of a changed readout with a discontinuity. Establishing the branch hypotheses for a given rigidity model remains an explicit obligation.
\end{proof}

\subsection{Structural Implications}

\begin{proposition}[The Mass Label Does Not Define Assignment Existence]
For fixed identity domain, codomain and readout family, the class defined by E0--E2 is independent of a numerical label $M_\Omega$. A claim that $M_\Omega=0$ excludes assignments needs an additional link to the domain or admissible readouts.
\end{proposition}

\begin{proof}
The displayed definition of $\mathcal P(M_\Omega)$ tests E0--E2, none of which contains that scalar. For an empty trajectory domain E2 fails, since a function on a finite cut would agree vacuously on every trajectory. Deriving such domain loss from $M_\Omega=0$ is not one of E0--E2 and cannot be inserted without a model.
\end{proof}

\begin{proposition}[Deformable Branch with Comparison Data]
For $0<M_\Omega<\infty$, the positive-displacement and lower-bound hypotheses of Theorem~\ref{thm:fragmentation} imply fragmentability.
\end{proposition}

\begin{proof}
Apply the quantitative inequality of Theorem~\ref{thm:fragmentation} to its witnessing finite-cost intervention.
\end{proof}

\begin{proposition}[Rigid Branch with Comparison Data]
For $M_\Omega=+\infty$, continuous finite readouts under E1 imply continuity, and the zero-displacement branch implies perturbation invariance. These are the respective hypotheses of Theorem~\ref{thm:forced-continuity}.
\end{proposition}

\begin{proof}
Apply the two conclusions of Theorem~\ref{thm:forced-continuity} with their separate hypotheses.
\end{proof}

\section{Extended Proofs}

\subsection{Proof of Fragmentation Theorem (Extended)}

\begin{proof}[Full Proof of Theorem~\ref{thm:fragmentation}]
Use the specified common space $Z$, map $F$ and finite-cost transport $h_\varepsilon$. Let $x$ be its positive-displacement witness and put $\delta=d_Z(x,h_\varepsilon x)>0$. Apply the lower bound on that same common domain to obtain readout displacement at least $C\delta$. No comparison of set distances is substituted for this pointwise distance. If both readouts are estimated with errors at most $\eta$, the triangle inequality gives estimated separation at least $C\delta-2\eta$; it remains positive when $C\delta>2\eta$. This is a bound on pair separation, not a statement that a particular endpoint lies outside a null ball.
\end{proof}

\subsection{Proof of Forced Continuity Theorem (Extended)}

\begin{proof}[Full Proof of Theorem~\ref{thm:forced-continuity}]
For a state $x$ and error $\epsilon>0$, choose $n$ with $2A_n<\epsilon/2$. Continuity of $g_n$ gives a neighborhood of $x$ where $d_{\mathcal E}(g_n(x),g_n(y))<\epsilon/2$. The two tail errors and the finite-readout difference bound the limiting difference by $\epsilon$. Independently, a zero-displacement comparison satisfies $F(h_\varepsilon x)=F(x)$ exactly. Setwise invariance of the identity object alone would not imply either of these steps.
\end{proof}

\subsection{Structural Necessity Lemma}

\begin{lemma}[Inheritance under Audited Fixed Transport]
An assignment satisfying the zero-displacement comparison branch inherits invariance under that declared intervention class. This applies to CCR realizations only after checking that branch.
\end{lemma}

\begin{proof}
The comparison readout is the restriction of $F$, and every specified transport fixes its argument. Therefore its value is unchanged. This proves the restricted inheritance statement without assuming that scalar rigidity determines arbitrary assignments.
\end{proof}
