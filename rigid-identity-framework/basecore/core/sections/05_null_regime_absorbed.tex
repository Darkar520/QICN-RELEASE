\part{Null-Regime Structural Exclusion}
\section{Compatibility, Witnesses, and Non-Nullity}

\subsection{Compatibility Operator}
\begin{definition}[Compatibility Operator]\label{def:compatibility-operator}
Declare a complete metric comparison domain $(Z,d_Z)$, a complete metric poset $(\E,\preceq,\dE)$ with bottom $\nullphi$, and maps $\Phi_S:D_S\to\E$. The audited readout $\Phi:Z\to\E$ is defined on its declared pair class $W\subseteq Z\times Z$. Extension and isomorphism transports $j:D_S\to D_{S'}$ satisfy $\Phi_{S'}\circ j=\Phi_S$ when claimed. On $W$, a certified $C>0$ satisfies
\[
\dE(\Phi(x),\Phi(y))\geq C d_Z(x,y).
\]
One may choose $Z=\I$ for history comparisons. Comparing whole identity objects requires a separate declared embedding/metric; $\Phi(\I)$ is not a point value when the domain is the set $\I$. A separated extension witness supplies two distinct defined points in a common comparison domain, not merely a relabeled copy under exact transport. CCR does not supply such a pair or lower constant.
\end{definition}

\begin{definition}[Phenomenological Attractor Set]\label{def:phenomenological-attractor}
Use $\Attr(S)$ for structurally stable readout values under a declared perturbation/extension family, as defined below. Temporal attraction would additionally require a dynamics and a basin; neither is implied by this notation.
\end{definition}

\begin{definition}[Phenomenological Stability]\label{def:phenomenological-stability}
A value $e=\Phi_S(z)$ is structurally stable if a finite modulus $\omega(\rho)\to0$ as $\rho\downarrow0$ controls $\dE(\Phi_{S'}(Pz),e)\leq\omega(\rho)$ for all declared perturbation transports $P$ of cost at most $\rho$. Exact persistence instead requires equality. Stability does not force equality at a nonzero perturbation budget.
\end{definition}

\begin{lemma}[Bridge Lemma]\label{lem:null-bridge}
For any defined $e\in\E$, distance $\dE(e,\nullphi)=0$ implies $e=\nullphi$. No CCR assumption is needed.
\end{lemma}
\begin{proof}
This is metric separation.
\end{proof}

\subsection{Witness-Relative Null Separation}
\begin{theorem}[Conditional Null Separation]\label{thm:phenomenological-instability}
Suppose $(z_0,z_1)\in W$, $d_Z(z_0,z_1)=\epsilon>0$, and the lower constant in Definition~\ref{def:compatibility-operator} is $C>0$. Their images cannot both equal null, and
\[
\max_{i=0,1}\dE(\Phi(z_i),\nullphi)\geq C\epsilon/2.
\]
The historical instability label denotes joint exact null exclusion here, not dynamical instability of an individual null endpoint.
\end{theorem}
\begin{proof}
Joint nullity would give mutual image distance zero, contradicting $C\epsilon>0$. The triangle inequality bounds that mutual distance by the sum of their distances to null, at most twice their maximum.
\end{proof}

\begin{corollary}[Conditional Forced Non-Nullity]\label{cor:forced-non-nullity}
Under the separated-pair hypotheses, at least one witness image is non-null. If one endpoint is fixed at null, the other has null distance at least $C\epsilon$. CCR alone does not select a non-null endpoint or guarantee a stable regime.
\end{corollary}
\begin{proof}
The first assertion follows from joint null exclusion. In the anchored case, mutual image distance is the other endpoint's distance to null, so the lower bound applies directly.
\end{proof}

\subsection{Positive Regime Structure}
\begin{definition}[Positive Regime Space]\label{def:positive-regime-space}
Let $\Eplus=\E\setminus\{\nullphi\}$ with inherited order. For a witness pair, let $A=\{\Phi(z_0),\Phi(z_1)\}\setminus\{\nullphi\}$ and $A_0=A\cup\{\nullphi\}$.
\end{definition}

\begin{theorem}[Minimal Positive Witness Regime]\label{thm:minimal-positive-regime}
The finite positive witness image $A$ is non-empty and has a minimal element. If its two possible values are comparable, it has a unique least element. No ambient order completeness is needed. This does not prove a least member of all of $\Eplus$.
\end{theorem}
\begin{proof}
Non-emptiness follows from joint null exclusion. A strictly descending chain in a finite non-empty poset terminates at a minimal element. If at most two members are comparable, the smaller is the unique least member. In the complete lattice $\{0,a,b,1\}$, incomparable positive $a,b$ are both minimal in their two-point image and have infimum zero. Thus an ambient infimum alone gives neither a positive infimum nor uniqueness.
\end{proof}

\begin{definition}[Structural Intensity]\label{def:structural-intensity}
Distinguish null distance $r(e)=\dE(e,\nullphi)$, global lower gap $\iota(e)=\inf_{f\in\E,\ f\prec e}\dE(e,f)$, and the audited relative gap $\iota_{A_0}(e)$ obtained by restricting this infimum to $f\in A_0$. All are measured in units of $\dE$. A positive null distance does not imply a positive global gap.
\end{definition}

\begin{theorem}[Anchored Witness Intensity Bound]\label{thm:universal-intensity-bound}
If one endpoint is null, the other image $e$ is the unique positive member of $A$ and satisfies $\iota_{A_0}(e)=r(e)\geq C\epsilon>0$. Without that anchor, only $\max_i r(\Phi(z_i))\geq C\epsilon/2$ follows. No universal global intensity bound is claimed.
\end{theorem}
\begin{proof}
Apply the anchored corollary. The only smaller member of $A_0$ is null, so the relative infimum equals null distance. On the usual interval $[0,1]$, the global gap at $1$ is zero despite relative gap one in $\{0,1\}$, showing why the restriction matters.
\end{proof}

\begin{theorem}[Conditional Phenomenological Closure]\label{thm:forced-phenomenological-closure}
A declared detected pair gives joint null exclusion and a minimal positive member of its finite image. A null anchor gives a unique positive witness value and the anchored quantitative bound. Uniqueness in the ambient regime space, temporal stability and CCR realizability require further hypotheses and constructions.
\end{theorem}
\begin{proof}
Combine the preceding witness, order and relative-gap results with their stated hypotheses. None of these implications supplies the premises of another by assuming the desired conclusion.
\end{proof}

\section{Canonical Families and Structural Closure Consequences}
\begin{proposition}[Profinite Bound]\label{prop:profinite-bound}
A profinite witness pair with declared positive separation $\epsilon$, certified lower constant $C_{\mathrm{prof}}>0$ and one null endpoint gives $r(e)\geq C_{\mathrm{prof}}\epsilon$. Summable coordinate costs do not by themselves certify CCR.
\end{proposition}
\begin{proof}
Apply the anchored bound. A transformation changing infinitely many coordinates can still cost at most the finite sum of the coordinate weights, so the old finite-support argument cannot establish infinite mass.
\end{proof}

\begin{proposition}[Symbolic Bound]\label{prop:symbolic-bound}
The same anchored conclusion holds for a symbolic witness with its audited metric and lower constant. The full binary shift with complement intervention and summable coordinate-change costs is a control against automatic infinite mass.
\end{proposition}
\begin{proof}
Apply the anchored bound. Complement changes every symbol at finite total summable cost; when admitted it yields positive deformation and finite infimum mass.
\end{proof}

\begin{theorem}[Rigidity Inheritance]\label{thm:rigidity-inheritance}
For $\Phi'=b\circ\Phi\circ a$, lower constants $c_a$ for $a$, $C$ for $\Phi$ on the mapped pairs and $c_b$ for $b$ give lower constant $c_ac_bC$. CCR inheritance separately requires every new finite-cost positive deformation to pull back to one in the old sector. Summability alone proves neither statement.
\end{theorem}
\begin{proof}
Compose the three inequalities. A new finite-cost positive deformation would contradict original CCR under the stipulated pullback property. Null anchors and stability moduli need their own transport conditions.
\end{proof}

\begin{theorem}[Factorization on a Faithful Image]\label{thm:null-categorical-universality}
If $\Phi:Z\to\E$ has a positive global lower bound, then every map $\Psi:Z\to F$ factors uniquely through $\Phi(Z)$. If $\Psi$ is $L$-Lipschitz, the factor is $L/C$-Lipschitz. No unique extension to the whole ambient $\E$ is implied.
\end{theorem}
\begin{proof}
The lower bound makes $\Phi$ injective with inverse Lipschitz constant $1/C$ on its image. Set $u(\Phi(z))=\Psi(z)$; this is well-defined, unique there, and has the stated bound. Outside the image no value is prescribed.
\end{proof}

\begin{theorem}[Finite-Prefix Simulation Bound]\label{thm:null-simulation-lower-bound}
For $F(b)=\sum_{j\geq0}2b_j/3^{j+1}$ and a decoder observing only bits $0,\ldots,N$, the optimal worst-case scalar error is $1/(2\,3^{N+1})$. It is positive for every fixed prefix and tends to zero with growing budget. The conclusion concerns prefix information, not finite real dimension.
\end{theorem}
\begin{proof}
Two extreme continuations of the same prefix have images separated by $3^{-(N+1)}$, forcing half that error for any single decoder value. The midpoint attains the bound on every fiber. This also supplies a non-empty approximation model consistent with revised E1/E2.
\end{proof}

\begin{theorem}[Conditional Closure by Non-Alternatives]\label{thm:null-closure-by-non-alternatives}
Any alternative model satisfying the same declared pair, lower-bound, anchor and transport hypotheses obeys the corresponding preceding conclusions. The program of realizing these properties jointly with an infinite-mass intervention model remains open; pair separation is not a proof of an individual null state's dynamical instability.
\end{theorem}
\begin{proof}
The conclusions depend on the enumerated mathematical hypotheses and apply to every model satisfying them. Existence and physical interpretation are additional questions.
\end{proof}
