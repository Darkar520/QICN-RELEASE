\part{Discrete-to-Continuous Bridge}
\section{From Node Networks to Hilbert Space Dynamics}

\noindent\textbf{Boundary of this bridge.}\enspace This part states a conditional approximation theorem between a finite node-network transition and the abstract BaseCore transition operator from Definition~\ref{def:transition}. It does not prove that any particular runtime execution belongs to an operational-consciousness class, and it does not convert finite tests into external validation.

\begin{definition}[Discrete State Space]
Let $N\geq1$ and let a node network be indexed by $i\in\{1,\dots,N\}$. Each node carries:
\begin{itemize}[leftmargin=*]
\item a local coherence state $x_i(t)\in[0,1]$;
\item an energy coordinate $e_i(t)\in[0,1]$;
\item a phase coordinate $\phi_i(t)\in[0,2\pi)$.
\end{itemize}
The discrete state space is
\[
\mathcal{S}_N=[0,1]^N\times[0,1]^N\times[0,2\pi)^N.
\]
\end{definition}

\begin{definition}[Mean Coherence Observable]
For a discrete state $s(t)=((x_i(t))_i,(e_i(t))_i,(\phi_i(t))_i)\in\mathcal{S}_N$, define
\[
X_N(t):=\frac{1}{N}\sum_{i=1}^N x_i(t)\in[0,1].
\]
This is an observable of the coherence coordinate only; it is not a complete state descriptor.
\end{definition}

\begin{definition}[Finite Hilbert Embedding]
Let
\[
\Hilb_N:=\mathbb{R}^N,\qquad
\inner{\mathbf{a}}{\mathbf{b}}_N:=\frac{1}{N}\sum_{i=1}^N a_i b_i.
\]
The coherence embedding
\[
\iota_N:\mathcal{S}_N\to\Hilb_N
\]
is
\[
\iota_N((x_i)_i,(e_i)_i,(\phi_i)_i)=(x_1,\dots,x_N).
\]
Thus $\iota_N$ forgets energy and phase except where they enter the transition rule below.
\end{definition}

\begin{definition}[Piecewise-Constant Isometry]
Let $\Hilb=L^2([0,1])$ and partition $[0,1]$ into intervals
\[
I_{N,i}=\left[\frac{i-1}{N},\frac{i}{N}\right),\qquad i=1,\dots,N.
\]
Define $J_N:\Hilb_N\hookrightarrow \Hilb$ by
\[
(J_N\mathbf{x})(s)=x_i\quad\text{for }s\in I_{N,i}.
\]
Then $J_N$ is an isometry from $(\Hilb_N,\norm{\cdot}_N)$ into $L^2([0,1])$. Its adjoint $J_N^\dagger:\Hilb\to\Hilb_N$ is the cell-averaging map
\[
(J_N^\dagger f)_i=N\int_{I_{N,i}}f(s)\,ds.
\]
\end{definition}

\begin{definition}[Discrete Transition Operator]
Fix auxiliary energy and phase coordinates $(e_i)_i$ and $(\phi_i)_i$, a graph with incoming neighborhoods $\operatorname{conn}(i)$, and a control $u\in\U$. Write
\[
C_N=[0,1]^N,\qquad
d_N^{\rm in}=\max_i|\operatorname{conn}(i)|,\qquad
d_N^{\rm out}=\max_j|\{i:j\in\operatorname{conn}(i)\}|.
\]
The coherence-coordinate transition
\[
\widehat T_{N,u}^{e,\phi}:C_N\to C_N
\]
is defined componentwise by
\[
\bigl(\widehat T_{N,u}^{e,\phi}\mathbf{x}\bigr)_i
=
\operatorname{clamp}_{[0,1]}\!\left(
x_i+\eta R_i(\mathbf{x};e,\phi)+\alpha\,B_i(\mathbf{x})+\Gamma_{N,i}(u)
\right),
\]
where
\[
R_i(\mathbf{x};e,\phi)
=
\sum_{j\in\operatorname{conn}(i)}
I_j(x_j)
\frac{\exp(-|e_i-e_j|/(kT))}
{\theta\cos(\phi_i-\phi_j)}.
\]
Here $I_j:[0,1]\to[0,1]$ is the intensity readout, $B_i$ is a feedback term, $\Gamma_N(u)$ is the finite input coupling, and $k,T,\theta>0$, $\eta,\alpha\geq0$ are parameters. Energy and phase are frozen in this operator; contractivity of this coordinate map does not establish contractivity of a coupled energy--phase evolution.
\end{definition}

\begin{assumption}[Discrete Regularity Conditions]\label{ass:discrete-regularity}
For a fixed parameter region, assume:
\begin{enumerate}[leftmargin=*]
\item $I_j$ is $L_I$-Lipschitz for every node $j$.
\item The phase denominator is uniformly separated from zero: there exists $c_\phi>0$ such that $|\cos(\phi_i-\phi_j)|\ge c_\phi$ on every active edge.
\item There exist $\beta>0$ and an $L_D$-Lipschitz map $D:C_N\to\Hilb_N$ such that $B(\mathbf{x})=-\beta\mathbf{x}+D(\mathbf{x})$.
\item $\Gamma_N(u)$ is independent of $\mathbf{x}$ for fixed $u$.
\item The parameters satisfy
\[
L_R:=\frac{L_I\sqrt{d_N^{\rm in}d_N^{\rm out}}}{\theta c_\phi},
\qquad
\rho_N:=|1-\alpha\beta|+\eta L_R+\alpha L_D<1.
\]
\end{enumerate}
\end{assumption}

\begin{remark}[Relation to the runtime guard]
A denominator guard can ensure that the quotient is defined on a parameter region. It does not establish the dissipative feedback condition or contractivity. On the stated energy domain and for $kT>0$, the exponential coefficient is at most $1$. The conditions above are sufficient, not necessary; a specified feedback must be checked against them.
\end{remark}

\begin{lemma}[Directed Resonance Bound]\label{lem:directed-resonance-bound}
Under the first two regularity conditions, $R$ is $L_R$-Lipschitz in the normalized Hilbert norm.
\end{lemma}

\begin{proof}
Let $a_{ij}$ be the energy--phase coefficient on an active edge and zero otherwise, and put $W=(|a_{ij}|)$. Componentwise,
\[
|R_i(\mathbf{x})-R_i(\mathbf{y})|
\leq L_I\sum_j W_{ij}|x_j-y_j|.
\]
The row and column sums of $W$ are bounded by $d_N^{\rm in}/(\theta c_\phi)$ and $d_N^{\rm out}/(\theta c_\phi)$. The matrix inequality $\norm{W}_2\leq\sqrt{\norm{W}_1\norm{W}_\infty}$ therefore gives the result. Multiplying both vector norms by $N^{-1/2}$ leaves the matrix bound unchanged. A bound on incoming degree alone does not control this norm for directed graphs.
\end{proof}

\begin{theorem}[Discrete Operator Contractivity]\label{thm:discrete-contraction}
Under Assumption~\ref{ass:discrete-regularity}, for fixed $u,e,\phi$ the operator $\widehat T_{N,u}^{e,\phi}$ is a strict contraction on the complete set $C_N$:
\[
\norm{\widehat T_{N,u}^{e,\phi}(\mathbf{x})-\widehat T_{N,u}^{e,\phi}(\mathbf{y})}_N
\le
\rho_N\norm{\mathbf{x}-\mathbf{y}}_N
\qquad
\forall \mathbf{x},\mathbf{y}\in C_N.
\]
It has a unique fixed point in $C_N$, and every iterate starting there converges to it.
\end{theorem}

\begin{proof}
The expression before clamping is
\[
(1-\alpha\beta)\mathbf{x}+\eta R(\mathbf{x})+\alpha D(\mathbf{x})+\Gamma_N(u).
\]
Subtract its values at two states, use Lemma~\ref{lem:directed-resonance-bound} and the Lipschitz bound for $D$, and apply non-expansiveness of the coordinate clamp. This gives the stated factor $|1-\alpha\beta|+\eta L_R+\alpha L_D$. The input cancels because $u$ is fixed. The cube is nonempty, closed and complete, and the clamp maps into it, so Banach's theorem applies.
\end{proof}

\begin{example}[Zero-Increment Negative Control]\label{ex:discrete-identity-control}
If $R=B=\Gamma_N=0$, the written transition is the identity on $C_N$ and has Lipschitz constant $1$ for $N\geq1$. It is not a strict contraction. In a decomposition $B=-\beta\operatorname{Id}+D$, one has $D=\beta\operatorname{Id}$, so $|1-\alpha\beta|+\alpha\beta\geq1$ and the sufficient conditions correctly reject it. Lipschitz regularity of increments alone cannot remove the identity term.
\end{example}

\begin{assumption}[Bridge Consistency]\label{ass:bridge-consistency}
Let $T_u=\aleph(K\cdot+\Gamma(u))$ be the BaseCore transition on $\Hilb=L^2([0,1])$, with $\kappa:=\norm{K}<1$. Define the closed invariant-domain candidate
\[
C=\{f\in L^2([0,1]):0\leq f\leq1\text{ almost everywhere}\}.
\]
Assume $T_u(C)\subseteq C$ and let $F_{N,u}=J_N\widehat T_{N,u}^{e,\phi}J_N^\dagger:C\to C$. There is a stated nonnegative bound $\eps_N$ such that
\[
\sup_{f\in C}\norm{F_{N,u}(f)-T_u(f)}_{\Hilb}\leq\eps_N.
\]
This is a discrepancy between nonlinear maps on a specified domain, not a linear operator norm. Its validity and any claimed rate must be proved for the discretization under consideration; H1--H5 alone do not provide a quadrature rate.
\end{assumption}

\begin{theorem}[Hilbert--Discrete Bridge]\label{thm:bridge}
Assume Hypotheses~\ref{hyp:H1}--\ref{hyp:H4} and Assumptions~\ref{ass:discrete-regularity} and~\ref{ass:bridge-consistency}. For $f_0\in C$ and $\mathbf{x}_0\in C_N$, set $f_{n+1}=T_u(f_n)$ and $\mathbf{x}_{n+1}=\widehat T_{N,u}^{e,\phi}(\mathbf{x}_n)$ with fixed $u,e,\phi$. Then, for every $n\geq0$,
\[
\norm{J_N\mathbf{x}_n-f_n}_{\Hilb}
\leq\kappa^n\norm{J_N\mathbf{x}_0-f_0}_{\Hilb}
+\eps_N\frac{1-\kappa^n}{1-\kappa}.
\]
The convention $\kappa^0=1$ applies also at $\kappa=0$. If $\eps_N\to0$, the transition maps converge uniformly on $C$.
\end{theorem}

\begin{proof}
Since $J_N^\dagger J_N=\operatorname{Id}$, $J_N\mathbf{x}_{n+1}=F_{N,u}(J_N\mathbf{x}_n)$. Insert $T_u(J_N\mathbf{x}_n)$, apply the uniform discrepancy and contractivity of $T_u$, and obtain $d_{n+1}\leq\eps_N+\kappa d_n$. Induction gives the geometric-sum bound. Uniform convergence is the definition of the discrepancy bound with $\eps_N\to0$.
\end{proof}

\begin{corollary}[Convergence of Fixed Points]\label{cor:bridge-fixed-points}
Let $f_u^\ast$ and $\widehat f_{N,u}^\ast$ be the unique fixed points of $T_u$ and $\widehat T_{N,u}^{e,\phi}$. Under the assumptions of Theorem~\ref{thm:bridge},
\[
\norm{J_N\widehat f_{N,u}^\ast-f_u^\ast}_{\Hilb}
\le
\frac{\eps_N}{1-\kappa}.
\]
Consequently, for a fixed $\kappa<1$, if $\eps_N\to0$, then
\[
\lim_{N\to\infty}\norm{J_N\widehat f_{N,u}^\ast-f_u^\ast}_{\Hilb}=0.
\]
\end{corollary}

\begin{proof}
Invariance of $C$ and completeness give $f_u^\ast\in C$. Write $g_N=J_N\widehat f_{N,u}^\ast\in C$. Then $F_{N,u}(g_N)=g_N$ and
\[
\norm{g_N-f_u^\ast}\leq\eps_N+\kappa\norm{g_N-f_u^\ast}.
\]
Rearrange. No uniform gap for the discrete contraction constants $\rho_N$ is needed for this estimate; each finite fixed point must exist, and the continuum factor $\kappa$ supplies the denominator.
\end{proof}

\begin{lemma}[Cell-Average Error for a Lipschitz Profile]\label{lem:cell-average-error}
Let $a:[0,1]\to\mathbb{R}$ be $L_a$-Lipschitz and put $P_N=J_NJ_N^\dagger$. Then
\[
\norm{P_Na-a}_{L^2}\leq\frac{L_a}{\sqrt{12}\,N}.
\]
\end{lemma}

\begin{proof}
For a cell $I$ of length $h=1/N$ and its mean $\bar a_I$, the variance identity gives
\[
\int_I|a-\bar a_I|^2
=\frac{1}{2h}\int_I\int_I|a(s)-a(t)|^2\,ds\,dt
\leq\frac{L_a^2h^3}{12}.
\]
Sum over $N$ cells and take square roots. For an affine profile the bound is attained with $L_a$ equal to its absolute slope.
\end{proof}

\begin{theorem}[Realized Contractive Bridge with Nonconstant Equilibria]\label{thm:realized-nonconstant-bridge}
Let $0<\kappa<1$, let $\U$ be a nonempty compact interval, and let a nonconstant Lipschitz profile $a:[0,1]\to\mathbb{R}$ satisfy
\[
0\leq a(s)+u\leq1-\kappa\qquad(s\in[0,1],\ u\in\U).
\]
Take $\I=C$, $Kf=\kappa\int_0^1f$, and $\Gamma(u)(s)=a(s)+u$. Let $a_{N,i}=N\int_{I_{N,i}}a$ and
\[
\widehat T_{N,u}(\mathbf{x})_i
=\operatorname{clamp}_{[0,1]}\left(\kappa\frac1N\sum_jx_j+a_{N,i}+u\right).
\]
This finite transition has the original increment form: choose all $N$ nodes as incoming neighbors (including self-edges), $I_j(x_j)=x_j$, $e_i=\phi_i=0$, $k=T=1$, $\theta=N/\kappa$, $\eta=\alpha=\beta=1$, $D=0$, $B(\mathbf{x})=-\mathbf{x}$ and $\Gamma_{N,i}=a_{N,i}+u$. The assumptions of the discrete contraction and bridge theorems hold with $\rho_N=\kappa$ and
\[
\eps_N=\norm{P_Na-a}\leq\frac{L_a}{\sqrt{12}\,N}.
\]
Writing $\bar a=\int_0^1a$, the equilibria are
\[
f_u^\ast(s)=a(s)+u+\kappa\frac{\bar a+u}{1-\kappa},\qquad
\widehat f_{N,u,i}^\ast=a_{N,i}+u+\kappa\frac{\bar a+u}{1-\kappa}.
\]
In particular, H1--H5 hold and
\[
\norm{J_N\widehat f_{N,u}^\ast-f_u^\ast}
=\norm{P_Na-a}\leq\frac{L_a}{\sqrt{12}\,N},\qquad
\dist(f_u^\ast,\mathcal N)=\norm{a-\bar a}>0.
\]
These conclusions hold simultaneously for every $u\in\U$.
\end{theorem}

\begin{proof}
$C$ is nonempty, closed and convex in $L^2$, $\norm{K}=\kappa$, and $u\mapsto\Gamma(u)$ is Lipschitz on compact $\U$. The range condition makes the clamp inactive on $C$ and on $C_N$. The chosen graph has energy--phase coefficient $1/\theta=\kappa/N$ on each edge, so the increment $\mathbf{x}+R(\mathbf{x})+B(\mathbf{x})$ equals the constant vector $\kappa N^{-1}\sum_jx_j$. The contraction factor is therefore $\kappa$ in both spaces.

Cell averaging preserves the mean. Consequently $F_{N,u}(f)-T_u(f)=P_Na-a$ for every $f\in C$, proving the uniform discrepancy and its rate by Lemma~\ref{lem:cell-average-error}. Averaging either fixed-point equation gives $m=\kappa m+\bar a+u$, hence $m=(\bar a+u)/(1-\kappa)$ and the displayed formulas. Their difference is exactly $P_Na-a$. The orthogonal projection of $f_u^\ast$ onto the constant subspace is its mean, leaving $a-\bar a$ as residual. A nonconstant continuous profile has positive residual norm. Thus no equilibrium is constant, and uniqueness excludes a constant fixed point: H5 follows for this class rather than being an additional assumption about it.
\end{proof}

\begin{example}[Explicit Common Witness]\label{ex:common-bridge-witness}
Choose $\kappa=1/4$, $a(s)=s/4$ and $\U=[0,1/4]$. All preceding conditions hold. Directly,
\[
f_u^\ast(s)=\frac{s}{4}+\frac1{24}+\frac{4u}{3},\qquad
\norm{J_N\widehat f_{N,u}^\ast-f_u^\ast}
=\frac{1}{4\sqrt{12}\,N},\qquad
\dist(f_u^\ast,\mathcal N)=\frac1{4\sqrt{12}}.
\]
For $N\geq2$ the finite equilibrium coordinates are also nonconstant. This is a mathematical realization of the projection dynamics, dissipative finite transition, consistency and non-collapse obligations together. It does not establish the inverse-limit non-finite-determination condition, the six operational invariants or a phenomenal bridge.
\end{example}

\section{Projective Dissipation on the Circle-Prefix Towers}

Fix the two towers of Theorem~\ref{thm:joint-transport-readout}, with $w_t=3^{-t}$, jump $a$ at stage $k$, and branch $s\in\{0,1\}$. Write $\beta_k=3^{-(k+1)}$. A history is specified by an initial angle $\theta$ and an infinite bit sequence $b$, with stage-$t$ angle $\theta+sa\mathbf{1}_{t>k}$. The depth $t$ is distinct from the iteration index $n$. For a fixed parameter $u\in\mathbb T=\mathbb R/\mathbb Z$, put
\[
 R(b)_1=0,\qquad R(b)_{j+1}=1-b_j,\qquad
 D_{s,u,t}(\theta_t,b_1,\ldots,b_t)
 =\bigl(u+sa\mathbf{1}_{t>k},\,R(b)|_t\bigr).
\]
Here $R(b)|_0$ is empty, and the right side at depth $t$ depends only on the given prefix. Write $D_{s,u}$ for the corresponding evolution of complete histories.

\begin{theorem}[Contractive Projective Evolution and Equilibrium Transport]\label{thm:projective-dissipation}
The stage maps commute with their bonding maps and with the existing transport $h_t$ between towers. At fixed $u$, the map $D_{s,u}$ has sharp Lipschitz constant $1/3$ on $I_s$ in $d_w$. Its unique fixed history $x^\ast_{s,u}$ has initial angle $u$ and binary sequence $0101\ldots$. For $n\geq 0$ its orbit satisfies
\[
 d_w(D_{s,u}^{\,n}x,x^\ast_{s,u})\leq 3^{-n}d_w(x,x^\ast_{s,u}).
\]
The existing transport $H$ conjugates the two evolutions and sends $x^\ast_{0,u}$ to $x^\ast_{1,u}$. Their $d_w$ and common readout distances are both $\beta_k a$. The finite readouts retain E0--E2 on each history domain, with the original uniform tail $3^{-t}$.
\end{theorem}

\begin{proof}
Deleting the last bit commutes with taking the stated initial segment of $R(b)$. At stage $k$, subtracting $sa$ from the updated phase $u+sa$ gives $u$ as required. At every other stage the updated phase is the same on both sides of the bonding map. Transport adds $a$ precisely at $t>k$ and leaves the bits fixed, giving the transport square. The induced history equations follow coordinatewise.

For points in one tower the phase difference vanishes after one iteration. If the input bits first differ at $m$, the output bits first differ at $m+1$. The binary distance therefore changes from $3^{-m}$ to $3^{-(m+1)}$. Equality in the contraction bound is attained with equal input phases and distinct bits. The fixed bit sequence is forced by $b_1=0$ and $b_{j+1}=1-b_j$, and the fixed angle by the phase update. Iteration gives the claimed geometric estimate. The phase difference under $H$ is $a$ at levels $t>k$, and Theorem~\ref{thm:joint-transport-readout} gives the readout distance. The underlying domains and readout are the same as in that theorem. A later bit may still be flipped within each finite cylinder, which proves E2 on the history domain.
\end{proof}

\begin{theorem}[Hilbert Realization of the Projective Dissipation]\label{thm:hilbert-projective-dissipation}
The circle-prefix evolutions above embed into the BaseCore projected dynamics on $L^2([0,1])$ with $C=\{0\leq f\leq1\text{ a.e.}\}$, $\|K\|=1/3$, and compact parameter space $\{0,1\}\times\mathbb T$. The embedding is bi-Lipschitz on $I_0\cup I_1$ for fixed $k$. The equilibrium profiles $f^\ast_{s,u}$ satisfy
\[
 \inf_{s,u}\operatorname{dist}(f^\ast_{s,u},\mathcal N)\geq\sqrt{2/45},
\]
where $\mathcal N$ is the subspace of constant $L^2$ functions. Thus H5 follows for this family.

At depth $t$ let $Q_t$ be the orthogonal projection onto the retained profile cells, with zero tail. The realized finite map has $N_t=8\cdot 9^t$ equal-mesh nodes and its embedded update is $Q_tT_{s,u}$. It commutes with restriction in the sense $Q_tT_{s,u}=Q_tT_{s,u}Q_t$ on $C$. Its uniform one-step discrepancy on $C$ is at most $3^{-t}/\sqrt{2}$, with equality for $t\geq1$. Its fixed point is $Q_tf^\ast_{s,u}$, with error at most the same bound. Starting the finite evolution at $Q_tf_0$ gives this bound at every iteration $n\geq1$ relative to the orbit from $f_0$.
\end{theorem}

\begin{proof}
Partition $[0,1]$ up to null sets into four consecutive phase cells of length $1/8$ and consecutive binary cells $B_j$ of lengths $\ell_j=4\cdot 9^{-j}$ for $j\geq1$. These lengths sum to one. Encode each of $\theta_0,\theta_{k+1}$ by its pair
\[
 e(\theta)=\bigl((1+\cos(2\pi\theta))/2,\,(1+\sin(2\pi\theta))/2\bigr)
\]
on its two phase cells. Assign $b_j$ on $B_j$. The chord distance of the encoded angle pair is $\sin(\pi\rho)$ for circle distance $\rho\leq1/2$, hence lies between $2\rho$ and $\pi\rho$. At first differing bit $m$, the norm of the binary part lies between $2\cdot3^{-m}$ and $(3/\sqrt{2})3^{-m}$. Since $d_w$ on the union is the maximum of the initial angle distance, $\beta_k$ times the later angle distance, and the binary distance, these bounds give the claimed bi-Lipschitz property.

For $f\in L^2$, let $Kf$ be zero on the phase cells and $B_1$, and equal to the negative mean of $f$ on $B_j$ throughout $B_{j+1}$. Cauchy--Schwarz on each cell and $\ell_{j+1}/\ell_j=1/9$ give
\[
 \|Kf\|^2\leq\tfrac19\sum_{j\geq1}\int_{B_j}|f|^2\leq\tfrac19\|f\|^2.
\]
A nonzero function supported on and constant on $B_1$ attains equality. The forcing $\Gamma(s,u)$ assigns the phase values $e(u),e(u+sa)$, zero on $B_1$, and one on each later $B_j$. This is continuous on the compact parameter space. On $C$, the affine output lies in $C$, so projection onto $C$ leaves it unchanged. On the embedded histories, it produces exactly the two phase resets and the bit update, preserving each embedded tower for its branch parameter. H1--H4 hold. The unique fixed profile has value zero on $B_1$ and one on $B_2$. For any constant $c$, its squared distance to that constant is at least $\ell_1c^2+\ell_2(1-c)^2\geq\ell_1\ell_2/(\ell_1+\ell_2)=2/45$. This proves the bound and H5. The same two retained cells certify the lower bound in the finite embedding for $t\geq2$.

The retained cells have endpoints on the equal mesh of size $1/N_t$. The equal-mesh averaging of $f$ therefore preserves each retained cell mean. To implement the update on mesh nodes, take the resonance term zero and use the feedback $B(\mathbf x)=-\mathbf x+L_t\mathbf x$, with $\alpha=1$, where $L_t$ is the equal-mesh restriction of $Q_tK$ to piecewise constant inputs. The input is the mesh vector for $Q_t\Gamma(s,u)$. The increment $\mathbf x+B(\mathbf x)$ is $L_t\mathbf x$, whose normalized Euclidean operator norm is at most $1/3$ by the mesh isometry. The clamp is inactive on $C$ and the embedded update is $Q_tT_{s,u}$. Every retained output depends only on retained means or the parameter. Hence $Q_tT_{s,u}=Q_tT_{s,u}Q_t$ on $C$.

The difference $T_{s,u}f-Q_tT_{s,u}f$ is a tail supported on cells of total measure $\tfrac12\,9^{-t}$, with values in $[0,1]$. All tail values equal one for $f=0$ if $t\geq1$. For $t=0$, $B_1$ is forced to zero and the sharp discrepancy is $1/(3\sqrt{2})$, still within the stated upper bound $1/\sqrt{2}$. The projected fixed point solves the finite equation by commutation, and contraction ensures uniqueness. By induction the finite trajectory from $Q_tf_0$ equals $Q_tT_{s,u}^{\,n}f_0$. The same tail bound applies to $T_{s,u}^{\,n}f_0$ for every $n\geq1$.
\end{proof}

\begin{remark}[Controls for Projective Dissipation]\label{rem:projective-dissipation-controls}
Holding bits fixed preserves positive distances on equal-phase fibers, ruling out strict contraction in $d_w$. Replacing the complement by a plain right shift with leading zero keeps the contraction but changes the fixed sequence to $0000\ldots$. At $u=0$ on $I_0$ this recovers the original null readout $(0,0,0)$. With the complement, the fixed ternary readout is $\sum_{r\geq1}2\cdot3^{-2r}=1/4$, so each fixed readout is at least $1/4$ away from that null element. These are separate from H5, whose distance is measured in $L^2$. For $a>0$ the branch-preserving map on the union has two distinct fixed points and therefore cannot be a strict contraction there. Although E2 holds on the history domain, the restriction to a singleton equilibrium at a fixed parameter factors through every stage. The model dissipates dependence on the initial history, and its phase component resets in one step.
\end{remark}


\begin{example}[Contractivity Alone Does Not Give Uniform Discretization]\label{ex:no-uniform-discretization}
Let $T(f)=\kappa f$ on $C$, with $0<\kappa<1$, and $F_N(f)=\kappa P_Nf$. In each cell choose $f_N$ equal to $0$ on the first half and $1$ on the second. Then $P_Nf_N=1/2$ and
\[
\norm{F_N(f_N)-T(f_N)}=\kappa/2
\]
for every $N$. Thus even strict contractions need additional structure for a uniform approximation rate over all of $C$. The profile-based realization above supplies that structure explicitly.
\end{example}

\begin{remark}[Applicability to the Current QICN Runtime]\label{rem:bridge-applicability}
Application to any implementation requires identifying its actual state domain, coefficients, feedback, frozen or evolving auxiliary coordinates, and approximation error. Network size or mean degree alone cannot discharge these conditions. The realized family above is an explicit mathematical example; no assertion is made here that a current runtime implements it or satisfies its hypotheses.
\end{remark}
